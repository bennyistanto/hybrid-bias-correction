# =============================================================================

# ---- Option A: pip (works everywhere) ----
#   pip install -r requirements.txt
#
# ---- Option B: mamba/conda (recommended for geospatial packages) ----
#   Use environment.yml instead — it handles conda channels, ESMPy, and
#   CLI tools (nco, cdo) that pip cannot install:
#
#   mamba env create -f environment.yml
#   mamba activate biascorr
#   python -m ipykernel install --user --name biascorr --display-name "Python (biascorr)"
#
# ---- Option C: pip inside a mamba env ----
#   mamba create -n biascorr python=3.12
#   mamba activate biascorr
#   pip install -r requirements.txt
#
# Tested with Python 3.10–3.12 (recommended: 3.12)
# TensorFlow 2.19 supports Python 3.9–3.13
